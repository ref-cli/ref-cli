# To generate a self-signed certificate and key:
openssl req -x509 -newkey rsa:4096 -keyout key.pem -out cert.pem -days 365 -nodes

# To generate an RSA private key:
openssl genrsa -out private.pem 4096

# To view certificate details:
openssl x509 -in cert.pem -noout -text

# To check certificate expiry:
openssl x509 -in cert.pem -noout -enddate

# To check a remote certificate:
echo | openssl s_client -connect example.com:443 2>/dev/null | openssl x509 -noout -dates

# To create a SHA-256 hash of a file:
openssl dgst -sha256 file.txt

# To encode a file to base64:
openssl base64 -in file.bin -out file.b64

# To decode a base64 file:
openssl base64 -d -in file.b64 -out file.bin

# To create a certificate signing request (CSR):
openssl req -new -key private.pem -out request.csr

# To verify a certificate against a CA:
openssl verify -CAfile ca.pem cert.pem
