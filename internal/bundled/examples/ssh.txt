# To connect to a remote host:
ssh user@host

# To connect on a non-default port:
ssh -p 2222 user@host

# To run a command on a remote host:
ssh user@host "ls -la /var/log"

# To forward a local port to a remote host (local tunnel):
ssh -L 8080:localhost:80 user@host

# To forward a remote port to local (reverse tunnel):
ssh -R 9090:localhost:3000 user@host

# To copy SSH public key to a remote host:
ssh-copy-id user@host

# To generate an ED25519 key pair:
ssh-keygen -t ed25519 -C "your@email.com"

# To use a specific identity file:
ssh -i ~/.ssh/id_rsa user@host

# To start an SSH agent and add a key:
eval $(ssh-agent) && ssh-add ~/.ssh/id_ed25519

# To test SSH connection without executing a command:
ssh -T git@github.com

# To proxy through a jump host:
ssh -J jumpuser@jumphost user@targethost
